
#include <stdio.h>

int main()
{
    int n, first, last, temp, place = 1, result;

    scanf("%d", &n);

    last = n % 10;

    temp = n;

    while (temp >= 10)
    {
        temp = temp / 10;
        place = place * 10;
    }

    first = temp;

    result = last * place + (n % place - last) + first;

    printf("%d", result);

    return 0;
}
